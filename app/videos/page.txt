'use client'

import { useState, useEffect } from 'react'
import Navigation from '@/components/Navigation'
import VideoPlayer from '@/components/VideoPlayer'
import { CheckCircle, Play, Filter, Sparkles } from 'lucide-react'

const VIDEO_LIBRARY = [
  { id: 'v1', title: 'Overcoming Exam Anxiety: 3 Science-Backed Hacks', category: 'Exam Anxiety', duration: '3 min', language: 'English / Hindi', youtubeId: 'VZTGBj7Zyvk' },
  { id: 'v2', title: 'The Procrastination Loop & How to Break It', category: 'Procrastination', duration: '4 min', language: 'English', youtubeId: 'LGVl9u3JvTo' },
  { id: 'v3', title: 'Imposter Syndrome in College: You Belong Here', category: 'Imposter Syndrome', duration: '3 min', language: 'English / Hinglish', youtubeId: '00AXMsgb0gM' },
  { id: 'v4', title: 'Homesickness & Adjusting to Campus Life', category: 'Homesickness', duration: '3 min', language: 'Hindi', youtubeId: 'U8WU1vBvmtc' },
  { id: 'v5', title: 'Sleep Hygiene & Managing Blue Light Before Bed', category: 'Sleep & Screens', duration: '4 min', language: 'English', youtubeId: 'Quw6URDE_Oo' },
  { id: 'v6', title: 'Myth vs Fact: Mental Health Stigma in India', category: 'Stigma & Myths', duration: '7 min', language: 'English / Hindi', youtubeId: 'NiUu8mMZjEA' },
]

export default function VideosPage() {
  const [selectedVideo, setSelectedVideo] = useState(VIDEO_LIBRARY[0])
  const [selectedCategory, setSelectedCategory] = useState('All')
  const [watchedSet, setWatchedSet] = useState<Set<string>>(new Set())

  const categories = ['All', 'Exam Anxiety', 'Procrastination', 'Imposter Syndrome', 'Homesickness', 'Sleep & Screens', 'Stigma & Myths']

  const filteredVideos = selectedCategory === 'All'
    ? VIDEO_LIBRARY
    : VIDEO_LIBRARY.filter(v => v.category === selectedCategory)

  const handleVideoComplete = (id: string) => {
    setWatchedSet(prev => new Set(prev).add(id))
    // Optionally track completion to API
    fetch('/api/exercises', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ exerciseId: `video-${id}`, exerciseName: selectedVideo.title, category: 'video' }),
    }).catch(() => {})
  }

  return (
    <div className="min-h-screen pb-24" style={{ background: 'linear-gradient(180deg, #f0fdfa 0%, #f8fffe 100%)' }}>
      <Navigation />

      <main className="max-w-4xl mx-auto px-4 pt-4">
        <div className="mb-6">
          <h1 className="text-2xl font-bold text-gray-800">Micro-Learning Video Library</h1>
          <p className="text-gray-500 mt-1">Bite-sized, practical mental health advice for student life</p>
        </div>

        {/* Active Player Hero */}
        <div className="mb-8">
          <VideoPlayer
            youtubeId={selectedVideo.youtubeId}
            title={selectedVideo.title}
            category={selectedVideo.category}
            duration={selectedVideo.duration}
            language={selectedVideo.language}
            isWatched={watchedSet.has(selectedVideo.id)}
            onComplete={() => handleVideoComplete(selectedVideo.id)}
          />
        </div>

        {/* Filters */}
        <div className="flex items-center gap-2 overflow-x-auto pb-3 mb-6 scrollbar-hide">
          <Filter className="w-4 h-4 text-teal-600 flex-shrink-0" />
          {categories.map(cat => (
            <button
              key={cat}
              onClick={() => setSelectedCategory(cat)}
              className={`px-3 py-1.5 rounded-full text-xs font-semibold whitespace-nowrap transition-all ${selectedCategory === cat ? 'bg-teal-600 text-white' : 'bg-white text-gray-600 border border-gray-200 hover:bg-teal-50'}`}
            >
              {cat}
            </button>
          ))}
        </div>

        {/* Video Grid */}
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
          {filteredVideos.map(video => {
            const isSelected = selectedVideo.id === video.id
            const isWatched = watchedSet.has(video.id)

            return (
              <div
                key={video.id}
                onClick={() => setSelectedVideo(video)}
                className={`glass-card p-4 cursor-pointer transition-all border-2 group ${isSelected ? 'border-teal-500 bg-teal-50/50 shadow-md' : 'border-transparent hover:border-teal-200'}`}
              >
                <div className="flex items-start justify-between mb-2">
                  <span className="text-xs font-semibold text-teal-700 bg-teal-100 px-2.5 py-0.5 rounded-full">
                    {video.category}
                  </span>
                  {isWatched && (
                    <span className="text-xs text-teal-600 font-semibold flex items-center gap-1">
                      <CheckCircle className="w-3.5 h-3.5" /> Watched
                    </span>
                  )}
                </div>

                <h3 className="font-bold text-gray-800 text-sm mb-2 group-hover:text-teal-700 transition-colors">
                  {video.title}
                </h3>

                <div className="flex items-center justify-between text-xs text-gray-400">
                  <span>⏱️ {video.duration}</span>
                  <span>🌐 {video.language}</span>
                </div>
              </div>
            )
          })}
        </div>
      </main>
    </div>
  )
}
